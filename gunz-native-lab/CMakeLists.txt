cmake_minimum_required(VERSION 3.20)
project(GunzNativeLab VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(FetchContent)
set(BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(BUILD_GAMES OFF CACHE BOOL "" FORCE)
FetchContent_Declare(raylib GIT_REPOSITORY https://github.com/raysan5/raylib.git GIT_TAG 5.5)
FetchContent_MakeAvailable(raylib)
add_executable(gunz_native_lab src/main.cpp)
target_link_libraries(gunz_native_lab PRIVATE raylib)
if(MSVC)
  target_compile_options(gunz_native_lab PRIVATE /W4 /permissive-)
else()
  target_compile_options(gunz_native_lab PRIVATE -Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(gunz_simulation_tests src/simulation_tests.cpp)
  target_include_directories(gunz_simulation_tests PRIVATE src)
  if(MSVC)
    target_compile_options(gunz_simulation_tests PRIVATE /W4 /permissive-)
  else()
    target_compile_options(gunz_simulation_tests PRIVATE -Wall -Wextra -Wpedantic)
  endif()
  add_test(NAME simulation_tests COMMAND gunz_simulation_tests)
endif()
